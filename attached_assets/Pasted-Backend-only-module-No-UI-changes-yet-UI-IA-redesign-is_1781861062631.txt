export function computeSystemRating(lead: Lead): number {
  // PLACEHOLDER — replace with founder's actual criteria when defined
  let score = 1
  if (lead.estimatedValue && lead.estimatedValue > 0) score++
  if (lead.possessionTimeline) score++
  if (lead.calls?.length > 0) score++
  if (lead.meetings?.some(m => m.status === 'SCHEDULED')) score++
  return Math.min(score, 5)
}

G6 — PROJECTS / DELIVERY BACKEND (DESIGN-PHASE SCOPE ONLY)

Add to schema:

enum ProjectPhase {
  DESIGN
  TECHNICAL
  PRODUCTION
  SITE_EXECUTION
  HANDOVER
  COMPLETED
}

enum ProjectHealth {
  ON_TRACK
  AT_RISK
  DELAYED
}

enum CollectionStatus {
  DUE
  INVOICED
  COLLECTED
}

model Project {
  id                String   @id @default(cuid())
  projectCode       String   @unique  // P-2026-XXX, same counter pattern as leadId
  leadId            String   @unique
  lead              Lead     @relation(fields: [leadId], references: [id])
  designerId        String?
  designer          User?    @relation(fields: [designerId], references: [id])
  phase             ProjectPhase  @default(DESIGN)
  health            ProjectHealth @default(ON_TRACK)
  progressPercent   Int      @default(0)
  contractValue     Float
  outstandingAmount Float    @default(0)
  location          String?
  handoverTargetDate DateTime?
  createdAt         DateTime @default(now())
  updatedAt         DateTime @updatedAt
  npsResponses      NPSResponse[]
  collections       Collection[]
  attentionFlags    ProjectAttentionFlag[]
}

model Collection {
  id          String   @id @default(cuid())
  projectId   String
  project     Project  @relation(fields: [projectId], references: [id])
  milestone   String
  amount      Float
  status      CollectionStatus @default(DUE)
  dueDate     DateTime?
  collectedAt DateTime?
  createdAt   DateTime @default(now())
}

model ProjectAttentionFlag {
  id          String   @id @default(cuid())
  projectId   String
  project     Project  @relation(fields: [projectId], references: [id])
  category    String   // 'Delayed' | 'At Risk' | 'Blocked' | 'Urgent service'
  description String
  resolvedAt  DateTime?
  createdAt   DateTime @default(now())
}
